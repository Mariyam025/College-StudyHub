1. if Statement

The if statement is used to execute a block of code when a condition is true.

Syntax

if(condition)
{
    statement;
}

Example Program

#include <stdio.h>

int main()
{
    int age = 20;

    if(age >= 18)
    {
        printf("Eligible");
    }

    return 0;
}
Output:
Eligible

2. if-else Statement

The if-else statement executes one block when the condition is true and another block when it is false.

Example Program

#include <stdio.h>

int main()
{
    int number = 7;

    if(number % 2 == 0)
    {
        printf("Even");
    }
    else
    {
        printf("Odd");
    }

    return 0;
}
Output:
Odd

3. Nested if

A nested if means an if statement placed inside another if statement.

Example Program

#include <stdio.h>

int main()
{
    int age = 20;
    int id = 1;

    if(age >= 18)
    {
        if(id == 1)
        {
            printf("Allowed");
        }
    }

    return 0;
}
Output:
Allowed

4. else-if Ladder

An else-if ladder is used when there are multiple conditions to check.

Example Program

#include <stdio.h>

int main()
{
    int marks = 75;

    if(marks >= 90)
    {
        printf("A+");
    }
    else if(marks >= 75)
    {
        printf("A");
    }
    else if(marks >= 60)
    {
        printf("B");
    }
    else
    {
        printf("C");
    }

    return 0;
}
Output:
A

5. switch Statement

The switch statement is used to select one block from multiple choices based on an expression.

Syntax

switch(expression)
{
    case value1:
        statement;
        break;

    case value2:
        statement;
        break;

    default:
        statement;
}

Example Program

#include <stdio.h>

int main()
{
    int choice = 2;

    switch(choice)
    {
        case 1:
            printf("Java");
            break;

        case 2:
            printf("C Programming");
            break;

        case 3:
            printf("Python");
            break;

        default:
            printf("Invalid choice");
    }

    return 0;
}
Output:
C Programming

6. break Statement

The break statement terminates the nearest loop or switch statement immediately.

Example Program

#include <stdio.h>

int main()
{
    int i;

    for(i = 1; i <= 5; i++)
    {
        if(i == 3)
        {
            break;
        }

        printf("%d ", i);
    }

    return 0;
}
Output:
1 2

7. continue Statement

The continue statement skips the remaining statements of the current loop iteration and continues with the next iteration.

Example Program

#include <stdio.h>

int main()
{
    int i;

    for(i = 1; i <= 5; i++)
    {
        if(i == 3)
        {
            continue;
        }

        printf("%d ", i);
    }

    return 0;
}
Output:
1 2 4 5

8. while Loop

A while loop repeatedly executes a block while its condition remains true.

Syntax

while(condition)
{
    statement;
}

Example Program

#include <stdio.h>

int main()
{
    int i = 1;

    while(i <= 5)
    {
        printf("%d ", i);
        i++;
    }

    return 0;
}
Output:
1 2 3 4 5

9. do-while Loop

A do-while loop executes its body first and checks the condition afterwards.

Syntax

do
{
    statement;
}
while(condition);

Example Program

#include <stdio.h>

int main()
{
    int i = 1;

    do
    {
        printf("%d ", i);
        i++;
    }
    while(i <= 5);

    return 0;
}
Output:
1 2 3 4 5

10. for Loop

A for loop is used when the initialization, condition and update can be written together.

Syntax

for(initialization; condition; update)
{
    statement;
}

Example Program

#include <stdio.h>

int main()
{
    int i;

    for(i = 1; i <= 5; i++)
    {
        printf("%d ", i);
    }

    return 0;
}
Output:
1 2 3 4 5

11. Nested Loops

A nested loop is a loop inside another loop.

Example Program

#include <stdio.h>

int main()
{
    int i, j;

    for(i = 1; i <= 3; i++)
    {
        for(j = 1; j <= 3; j++)
        {
            printf("* ");
        }

        printf("\n");
    }

    return 0;
}
Output:
* * *
* * *
* * *

🔍 Loop Comparison

Loop Important Point
while Condition is checked before the loop body.
do-while Body executes before the condition is checked.
for Initialization, condition and update are written together.

⚡ Quick Revision

  • if: Checks a condition.
  • if-else: Chooses between two blocks.
  • Nested if: if inside another if.
  • else-if ladder: Checks multiple conditions.
  • switch: Selects from multiple cases.
  • break: Terminates loop/switch.
  • continue: Skips current iteration.
  • while: Entry-controlled loop.
  • do-while: Executes body before checking condition.
  • for: Common loop with initialization, condition and update.
  • Nested loop: Loop inside another loop.
